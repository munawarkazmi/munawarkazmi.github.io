\section{Featured Research}

\entry{legibility-bounds: certified limits on legible motion}{\href{https://github.com/munawarkazmi/legibility-bounds}{github.com/munawarkazmi/legibility-bounds}}
\sub{Sole author. Python, computational geometry, exact predicates. Preprint for IEEE RA-L, DOI \href{https://doi.org/10.5281/zenodo.21834956}{10.5281/zenodo.21834956}.}
\begin{itemize}
\item Proved a ceiling \textbf{no trajectory within the path budget can exceed}, not the clarity one search
happened to find; \textbf{32 of 32 certified intervals hold, zero violations}, widest gap 0.0567.
\item The same argument prices safety: \textbf{3 of 8 keep-out pairs certify a positive cost}, the other 5 certify
nothing and say so, which no comparison of two searches can establish.
\end{itemize}

\entry{plan-failure-bench}{\href{https://github.com/munawarkazmi/plan-failure-bench}{github.com/munawarkazmi/plan-failure-bench}}
\sub{Sole author. Python, PDDL, differential testing, CI. Working paper on Zenodo.}
\begin{itemize}
\item Designed \textbf{60 proof-carrying instructions} over two symbolic environments; every ground-truth label is
a mechanical proof re-verified by \textbf{548 CI tests} on each commit.
\item \textbf{Judge-free scoring}: a deterministic checker, differentially tested against an independent PDDL
toolchain, assigns one verdict per response; detection is never reported without its paired false positive count.
\item Ran an \textbf{18-run grid over four models}, plain and semantically obfuscated; the versioned obfuscation
caught and retired two artefacts of its own first token scheme.
\end{itemize}

\entry{Trust-calibrated HRI on the iCub humanoid, MSc thesis}{University of Hertfordshire}

\item Built a sim-to-real pipeline transferring verbal and non-verbal intent communication to physical hardware;
